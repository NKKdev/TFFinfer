132 template<typename T>
133 bool load_array_meta(const std::shared_ptr<FileLoader> &file_loader,
134 const std::shared_ptr<tff::core::model::ModelContext> &gguf_ctx, const std::string &key,
135 const size_t &n,
136 const bool is_array = false) {
137 if (is_array) {
138 std::vector<T> value;
139 try {
140 if (!file_loader->read(value, n)) {
141 return false;
142 }
143 } catch (std::length_error &) {
144 tff::log::Logger::error("%s: encountered length_error while reading value for key '%s'\n", __func__,
145 key.c_str());
146 return false;
147 } catch (std::bad_alloc &) {
148 tff::log::Logger::error("%s: encountered bad_alloc error while reading value for key '%s'\n",
149 __func__,
150 key.c_str());
151 return false;
152 }
153 gguf_ctx->_kv.insert(std::make_pair(key, value));
154 } else {
155 T value;
156 file_loader->read(value);
157 gguf_ctx->_kv.insert(std::make_pair(key, value));
158 }
159 return true;
160 };
