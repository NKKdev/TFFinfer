19 static constexpr void hash_combine(std::size_t &seed, const T &v) {
20 std::hash<T> hasher;
21 seed ^= hasher(v) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
22 }

24 template<typename T, typename... Rest>
25 static constexpr void hash_combine(std::size_t &seed, const T &v, const Rest &... rest) {

56 static inline float fp16_to_fp32(uint16_t h) {
57 uint32_t sign = (h >> 15) & 1;
58 uint32_t exp = (h >> 10) & 0x1F;
59 uint32_t mant = h & 0x3FF;
60 if (exp == 0x1F) {
61 // Inf or NaN
62 return (mant == 0) ? (sign ? -INFINITY : INFINITY) : NAN;
63 }
64
65 uint32_t f32_bits = (sign << 31) |
66 ((exp == 0 ? 0 : (exp + 127 - 15)) << 23) |
67 (mant << 13);
68 return *reinterpret_cast<float *>(&f32_bits);
69 } //
70 static inline uint16_t fp32_to_fp16(float f) {
71 union {
72 float f;
73 uint32_t u;
74 } u = {f};
75 uint32_t sign = (u.u >> 16) & 0x8000;
76 int32_t exp = ((u.u >> 23) & 0xff) - 127;
77 uint32_t mantissa = u.u & 0x7fffff;
78
79 if (exp < -24) {
80 return static_cast<uint16_t>(sign);
81 } else if (exp < -14) {
82 mantissa |= 0x800000;
83 mantissa >>= (-14 - exp) + 1;
84 return static_cast<uint16_t>(sign | (mantissa >> 13));
85 } else if (exp > 15) {
86 return static_cast<uint16_t>(sign | 0x7c00 | (mantissa ? 0x200 : 0));
87 } else {
88 return static_cast<uint16_t>(sign | ((exp + 15) << 10) | (mantissa >> 13));
89 }
90 }
