1819 std::vector<DataType> get_general_value(const std::string key_value,
1820 const std::shared_ptr<tff::core::model::ModelContext> &ctx) {
1821 std::vector<DataType> result;
1822
1823
1824 auto kv_it = ctx->_kv.find(key_value);
1825 if (kv_it == ctx->_kv.end()) {
1826 return result;
1827 }
1828
1829 const auto &kvval = kv_it->second;
1830
1831 std::visit([&result](const auto &val) {
1832 using T = std::decay_t<decltype(val)>;
1833
1834 if constexpr (std::is_same_v<T, std::vector<DataType> >) {
1835 result = val;
1836 } else if constexpr (is_vector_v<T>) {
1837 using U = typename T::value_type;
1838 for (const auto &item: val) {
1839 if constexpr (std::is_convertible_v<U, DataType>) {
1840 result.push_back(static_cast<DataType>(item));
1841 } else if constexpr (requires { try_convert<DataType>(item); }) {
1842 if (auto cvt = try_convert<DataType>(item)) {
1843 result.push_back(*cvt);
1844 }
1845 }
1846 }
1847 } else if constexpr (std::is_same_v<T, ModelContext::BasicType>) {
1848 std::visit([&result](const auto &inner) {
1849 using U = std::decay_t<decltype(inner)>;
1850 if constexpr (std::is_convertible_v<U, DataType>) {
1851 result.push_back(static_cast<DataType>(inner));
1852 } else if constexpr (requires { try_convert<DataType>(inner); }) {
1853 if (auto cvt = try_convert<DataType>(inner)) {
1854 result.push_back(*cvt);
1855 }
1856 }
1857 }, val);
1858 } else if constexpr (!is_vector_v<T> && !std::is_same_v<T, ModelContext::BasicType>) {
1859 if constexpr (std::is_convertible_v<T, DataType>) {
1860 result.push_back(static_cast<DataType>(val));

1862 if (auto cvt = try_convert<DataType>(val)) {
1863 result.push_back(*cvt);
1864 }
1865 }
