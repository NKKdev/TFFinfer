139 [[nodiscard]] inline int64_t get_bytes() const {
140 size_t nbytes = 0;
141 const size_t blck_size = type_traits_auto[_data_type]._blck_size;
142 if (blck_size == 1) {
143 for (int i = _n_dims - 1; i < _n_dims; ++i) {
144 nbytes += (this->_shape[i]) * this->_strides[i];
145 }
146 } else {
147 nbytes = this->_shape[0] * this->_strides[0] / blck_size;
148 for (int i = _n_dims - 1; i < _n_dims; ++i) {
149 nbytes += (this->_shape[i] - 1) * this->_strides[i];
150 }
151 }
152
153 return nbytes;

246 inline void stride_infer() {
247 if (this->_strides.empty()) {
248 tff::log::Logger::error("_strides is empty!!");
249 return;
250 }
251 _strides[0] = _type_size;
252 if (this->_strides.size() > 1) {
253 _strides[1] = _strides[0] * (_shape[0] / _blk_size);
254 }
255 for (int j = 2; j < _shape.size(); ++j) {
256 _strides[j] = _strides[j - 1] * _shape[j - 1];
257 }

422 template<typename... Args>
423 inline size_t compute_offset(Args... indices) const {
424 constexpr size_t num_indices = sizeof...(indices);
425 if (num_indices > _shape.size()) {
426 throw std::invalid_argument(
427 "Number of indices (" + std::to_string(num_indices) +
428 ") does not match tensor dimensions (" + std::to_string(_shape.size()) + ")"
429 );
430 }
431 std::array<size_t, num_indices> idxs{static_cast<size_t>(indices)...};
432 size_t offset = 0;
433 for (size_t i = 0; i < num_indices; ++i) {
434 offset += idxs[i] * _strides[i];
435 }
436
437 return offset;
438 }
