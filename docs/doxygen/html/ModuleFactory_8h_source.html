204 template<typename Base>
205 std::shared_ptr<Base> create_shared(const std::string &type, const ModuleKeyType &key) {
206 auto it = instance()->_typed_factories.find(type);
207 if (it != instance()->_typed_factories.end()) {
208 using MapType = std::unordered_map<ModuleKeyType, CreatorInfo<Base>, ModuleKeyHash, ModuleKeyEqual>;
209 try {
210 auto &creators = std::any_cast<MapType &>(it->second._creators);
211 auto creator_it = creators.find(key);
212 if (creator_it != creators.end()) {
213 const auto &info = creator_it->second;
214 if (info.policy == CreationPolicy::PROTOTYPE) {
215 return info.creator();
216 } else if (info.policy == CreationPolicy::SINGLETON) {
217 return instance()->get_singleton_instance<Base>(key, info.creator);
218 }
219 }
220 } catch (const std::bad_any_cast &e) {
221 tff::log::Logger::error("create_shared: bad_any_cast %s", e.what());
222 }
223 }
224
225 return nullptr;
226 }
