427 struct GraphNodeSemanticKey {
428 TffOpType _op_type;
429 std::string _params_fingerprint;
430
431 explicit GraphNodeSemanticKey(const std::shared_ptr<tff::core::graph::GraphNode> &node)
432 : _op_type(node->op_type()) {
433 auto params = node->get_builder()->extract_params();
434
435 std::vector<std::string> kv_pairs;
436 for (const auto &[key, value]: params) {
437 if (key == "out") {
438 continue;
439 }
440 std::string val_str;
441 std::stringstream ss;
442 if (value.type() == typeid(std::shared_ptr<tff::core::memory::Tensor>)) {
443 auto tensor = std::any_cast<std::shared_ptr<tff::core::memory::Tensor> >(value);
444 ss << tensor.get();
445 val_str = "Tensor@" + ss.str();
446 } else if (value.type() == typeid(int)) {
447 ss << std::any_cast<int>(value);
448 val_str = ss.str();
449 } else if (value.type() == typeid(float)) {
450 ss << std::any_cast<float>(value);
451 val_str = ss.str();
452 } else if (value.type() == typeid(std::string)) {
453 val_str = std::any_cast<std::string>(value);
454 } else if (value.type() == typeid(bool)) {
455 val_str = std::any_cast<bool>(value);
456 } else if (value.type() == typeid(int64_t)) {
457 val_str = std::to_string(std::any_cast<int64_t>(value));
458 }
459 kv_pairs.push_back(key + "=" + val_str);
460 }
461
462 std::sort(kv_pairs.begin(), kv_pairs.end());
463
464
465 for (const auto &kv: kv_pairs) {
466 _params_fingerprint += kv + ";";
467 }
468 }
469
470 bool operator==(const GraphNodeSemanticKey &other) const {
471 return _op_type == other._op_type &&
472 _params_fingerprint == other._params_fingerprint;
473 }
474 };
475}
