183 explicit FileMMap(const std::shared_ptr<FileLoader> &file_loader,
184 size_t prefetch = static_cast<size_t>(-1), bool numa = false) {
185 _size = file_loader->size();
186 const int fd = file_loader->file_id();
187 int flags = MAP_SHARED;
188 if (numa) { prefetch = 0; }
189
190 if (posix_fadvise(fd, 0, 0, POSIX_FADV_SEQUENTIAL)) {
191 tff::log::Logger::warning("warning: posix_fadvise(.., POSIX_FADV_SEQUENTIAL) failed: %s\n",
192 strerror(errno));
193 }
194 if (prefetch) { flags |= MAP_POPULATE; }
195
196 _addr = mmap(NULL, file_loader->size(), PROT_READ, flags, fd, 0);
197 if (_addr == MAP_FAILED) {
198 tff::log::Logger::error("mmap failed: %s", strerror(errno));
199 }
200
201 if (prefetch > 0) {
202 if (posix_madvise(_addr, std::min(file_loader->size(), prefetch), POSIX_MADV_WILLNEED)) {
203 tff::log::Logger::warning("warning: posix_madvise(.., POSIX_MADV_WILLNEED) failed: %s\n",
204 strerror(errno));
205 }
206 }
207 if (numa) {
208 if (posix_madvise(_addr, file_loader->size(), POSIX_MADV_RANDOM)) {
209 tff::log::Logger::warning("warning: posix_madvise(.., POSIX_MADV_RANDOM) failed: %s\n",
210 strerror(errno));
211 }
212 }
213
214 _mapped_fragments.emplace_back(0, file_loader->size());
215 }

242 inline void align_range(size_t *first, size_t *last, size_t page_size) {
243 size_t offset_in_page = *first & (page_size - 1);
244 size_t offset_to_page = offset_in_page == 0 ? 0 : page_size - offset_in_page;
245 *first += offset_to_page;
246
247 *last = *last & ~(page_size - 1);
248
249 if (*last <= *first) {
250 *last = *first;
251 }
252 }

254 inline void FileMMap::unmap_fragment(size_t &first, size_t &last) {
255 const int page_size = sysconf(_SC_PAGESIZE);
256 align_range(&first, &last, page_size);
257 const size_t len = last - first;
258
259 if (len == 0) {
260 return;
261 }
262
263 void *next_page_start = static_cast<uint8_t *>(_addr) + first;
264
265 if (munmap(next_page_start, len)) {
266 tff::log::Logger::warning("warning: munmap failed: %s\n", strerror(errno));
267 }
268
269 std::vector<std::pair<size_t, size_t> > new_mapped_fragments;
270 for (const auto &frag: _mapped_fragments) {
271 if (frag.first < first && frag.second > last) {
272 new_mapped_fragments.emplace_back(frag.first, first);
273 new_mapped_fragments.emplace_back(last, frag.second);
274 } else if (frag.first < first && frag.second > first) {
275 new_mapped_fragments.emplace_back(frag.first, first);
276 } else if (frag.first < last && frag.second > last) {
277 new_mapped_fragments.emplace_back(last, frag.second);
278 } else if (frag.first >= first && frag.second <= last) {
279 } else {
280 new_mapped_fragments.push_back(frag);
281 }
282 }
283 _mapped_fragments = std::move(new_mapped_fragments);
284 }

313 inline void FileLock::grow_to(size_t target_size) {
314 if (_failed_already) {
315 return;
316 }
317 size_t granularity = sysconf(_SC_PAGESIZE);
318 target_size = (target_size + granularity - 1) & ~(granularity - 1);
319 if (target_size > _size) {
320 if (raw_lock((uint8_t *) _addr + _size, target_size - _size)) {
321 _size = target_size;
322 } else {
323 _failed_already = true;
324 }
325 }
