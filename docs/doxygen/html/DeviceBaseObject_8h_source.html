57 if constexpr (is_gpu()) {
58 return std::is_same_v<T, float> ||
59 std::is_same_v<T, half> ||
60 std::is_same_v<T, double> ||
61 std::is_same_v<T, int32_t> ||
62 std::is_same_v<T, int64_t> ||
63 std::is_same_v<T, uint8_t>;

66 std::is_same_v<T, double> ||
67 std::is_same_v<T, int32_t> ||
68 std::is_same_v<T, int64_t> ||
