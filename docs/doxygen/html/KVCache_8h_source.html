36 class PageManager : public std::enable_shared_from_this<PageManager> {
37 public:
38 PageManager(const int &device_id,
39 const tff::core::memory::DataType data_type,
40 int total_pages, int _d_h, int _h_kv,
41 std::shared_ptr<tff::core::runtime::LLMMemManager> &mem_manager_ptr,
42 int page_size = PAGE_SIZE)
43 : _device_id(device_id), _total_pages(total_pages), _page_size(page_size), _n_d_h(_d_h), _n_h_kv(_h_kv),
44 _mem_manager_ptr(mem_manager_ptr) {
45 _pages.resize(total_pages);
46 int64_t kv_cache_buffer_size = 0;
47 for (int i = 0; i < total_pages; ++i) {
48 _pages[i] = std::make_shared<KVPage>();
49 std::array<int64_t, MAX_TENSOR_DIM> shapes = {
50 static_cast<int64_t>(_d_h), static_cast<int64_t>(_h_kv), static_cast<int64_t>(page_size), 1
51 };
52 _pages[i]->_k = std::make_shared<tff::core::memory::Tensor>(
53 data_type, memory::MemoryType::TFF_MEM_TYPE_KV_CACHE,
54 shapes);
55 kv_cache_buffer_size += _pages[i]->_k->get_bytes();
56 _pages[i]->_v = std::make_shared<tff::core::memory::Tensor>(
57 data_type, memory::MemoryType::TFF_MEM_TYPE_KV_CACHE,
58 shapes);
59 kv_cache_buffer_size += _pages[i]->_v->get_bytes();
60 _pages[i]->_n_tokens = 0;
61 _pages[i]->_is_used = false;
62 _free_list.push_back(i);
63 }
64 auto offset = this->_mem_manager_ptr->allocate_memory_offset(kv_cache_buffer_size, device_id,
65 memory::MemoryType::TFF_MEM_TYPE_WORKSPACE);
66 this->_mem_manager_ptr->release_memory(device_id, offset);
67 };

81 inline PageID allocate() {
82 std::lock_guard<std::mutex> lock(_mutex);
83 if (_free_list.empty()) return INVALID_PAGE_ID;
84 PageID id = _free_list.back();
85 _free_list.pop_back();
86 _pages[id]->_is_used = true;
87 return id;
88 }

94 inline void free(PageID id) {
95 std::lock_guard<std::mutex> lock(_mutex);
96 if (id >= 0 && id < (int) _pages.size() && _pages[id]->_is_used) {
97 _pages[id]->_is_used = false;
98 _free_list.push_back(id);
99 }

107 inline std::shared_ptr<tff::core::memory::Tensor> get_k(PageID id,
108 const std::shared_ptr<core::device::DeviceEvent> &event)
109 const {
110 std::lock_guard<std::mutex> lock(_mutex);
111 if (id != INVALID_PAGE_ID && id < static_cast<int>(_pages.size()) && _pages[id]->_is_used) {
112 if (_pages[id]->_k->get_buffer() == nullptr) {
113 const auto [fst, snd] = _mem_manager_ptr->allocate_memory(_pages[id]->_k->get_bytes(),
114 this->_device_id,
115 memory::MemoryType::TFF_MEM_TYPE_KV_CACHE,
116 event);
117 // tff::log::Logger::info(
118 // "k device_id: %d,memory type: %d, allocate offset: %lld end offset : %lld",
119 // _device_id, memory::MemoryType::TFF_MEM_TYPE_KV_CACHE, fst,
120 // fst + _pages[id]->_k->get_bytes());
121 _pages[id]->_k->set_buffer_data(snd, _pages[id]->_k->get_bytes(),
122 fst);
123 }
124
125 return _pages[id]->_k;
126 } else {
127 return nullptr;
128 }

232 inline bool append_token() {
233 if (_num_tokens == get_max_tokens()) {
234 PageID new_page = _page_manager->allocate();
235 if (new_page == INVALID_PAGE_ID) return false;
236 _page_table.push_back(new_page);
237 }
238 _num_tokens++;
239 return true;
240 }

355 LayerKVContext *get_context(int seq_id, int layer_id) {
356 const int key = make_key(seq_id, layer_id);
357 std::lock_guard<std::mutex> lock(global_mutex);
358
359 auto it = this->_seq_contexts.find(key);

364 auto ctx = std::make_unique<LayerKVContext>(seq_id, layer_id, this->_page_manager);
