37 inline void append_token(const int32_t token) {
38 if (!_is_finished) {
39 _tokens.push_back(token);
40 this->_current_pos = this->_tokens.size() - 1;
41 auto iter = std::ranges::find(_eos_token_ids, token);

44 }

46 if (iter_eog != _eog_token_ids.end()) {
47 _is_finished = true;
